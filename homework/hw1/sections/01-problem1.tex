\problem[10]{前 n 项和}
\label{sec:sum}
求 $1+2+\cdots+n$，并证明公式。
\begin{solution}
公式为 $\dfrac{n(n+1)}{2}$。
\begin{proof}
对 $n$ 归纳。$n=1$ 时两边都是 $1$。
假设对 $n=k$ 成立，则
\[
  1+\cdots+(k+1)=\frac{k(k+1)}{2}+(k+1)=\frac{(k+1)(k+2)}{2}.
\]
\end{proof}
\end{solution}
